Prepare Data

breaks <- c(-Inf, 0, 3, 5, 10, 15, Inf)
# fmt: skip
category_labels <- c(
  "0", "1 - 3", "4 - 5", "6 - 10", "11 - 15", "Above 15"
)
fill_scale <- brewer.pal(6, "PuRd") |> rev()
category_cut <- function(x) {
  x |>
    cut(
      breaks = breaks,
      labels = category_labels,
      include.lowest = TRUE,
      right = TRUE
    ) |>
    factor(
      ordered = TRUE,
      levels = category_labels
    )
}
plot_data <-
  data |>
  mutate(
    category = 0:19 |>
      sample(
        size = n(),
        replace = TRUE,
        prob = seq(20, 1)
      ) |>
      category_cut(),
    color = category |>
      as.character() |>
      recode_values(
        from = category |> levels(),
        to = fill_scale
      )
  )
